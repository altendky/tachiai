cmake_minimum_required(VERSION 3.22)
project(tachiai_openvpn_validation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OPENVPN3_SOURCE_DIR "" CACHE PATH "Pinned OpenVPN3 source")
set(ASIO_SOURCE_DIR "" CACHE PATH "Pinned standalone Asio source")
if(ANDROID)
  set(TACHIAI_OPENVPN_DEPENDENCY_PREFIX "" CACHE PATH "Pinned per-ABI static dependencies")
  if(NOT CMAKE_ANDROID_STL_TYPE STREQUAL "c++_static" OR NOT ANDROID_PLATFORM_LEVEL EQUAL 26)
    message(FATAL_ERROR "Native Android validation requires API 26 and static libc++")
  endif()
  foreach(dependency ssl crypto fmt lz4)
    if(NOT EXISTS "${TACHIAI_OPENVPN_DEPENDENCY_PREFIX}/lib/lib${dependency}.a")
      message(FATAL_ERROR "Pinned static ${dependency} dependency missing")
    endif()
    add_library(tachiai_${dependency} STATIC IMPORTED)
    set_target_properties(tachiai_${dependency} PROPERTIES
      IMPORTED_LOCATION "${TACHIAI_OPENVPN_DEPENDENCY_PREFIX}/lib/lib${dependency}.a")
  endforeach()
  set(TACHIAI_NATIVE_LIBRARIES tachiai_ssl tachiai_crypto tachiai_fmt tachiai_lz4)
else()
  find_package(OpenSSL 3 REQUIRED)
  find_package(fmt REQUIRED)
  find_library(LZ4_LIBRARY NAMES lz4 REQUIRED)
  find_package(Threads REQUIRED)
  set(TACHIAI_NATIVE_LIBRARIES OpenSSL::SSL OpenSSL::Crypto fmt::fmt ${LZ4_LIBRARY} Threads::Threads)
endif()
add_library(tachiai_openvpn STATIC adapter.cpp
  ${OPENVPN3_SOURCE_DIR}/client/ovpncli.cpp
  ${OPENVPN3_SOURCE_DIR}/openvpn/crypto/data_epoch.cpp)
target_include_directories(tachiai_openvpn SYSTEM PRIVATE ${OPENVPN3_SOURCE_DIR} ${ASIO_SOURCE_DIR}/asio/include)
if(ANDROID)
  target_include_directories(tachiai_openvpn SYSTEM PRIVATE ${TACHIAI_OPENVPN_DEPENDENCY_PREFIX}/include)
endif()
target_compile_definitions(tachiai_openvpn PRIVATE ASIO_STANDALONE USE_ASIO USE_OPENSSL HAVE_LZ4 USE_TUN_BUILDER OPENVPN_CORE_API_VISIBILITY_HIDDEN)
target_link_libraries(tachiai_openvpn PUBLIC ${TACHIAI_NATIVE_LIBRARIES})
target_compile_options(tachiai_openvpn PRIVATE -Wall -Wextra)
set_target_properties(tachiai_openvpn PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  # Complete native link proof only; this shared object is not an app artifact.
  add_library(tachiai_openvpn_linkcheck SHARED linkcheck.cpp)
  target_link_libraries(tachiai_openvpn_linkcheck PRIVATE tachiai_openvpn)
  target_link_options(tachiai_openvpn_linkcheck PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined -Wl,--exclude-libs,ALL)
else()
  add_executable(tachiai_openvpn_test adapter_test.cpp)
  target_compile_definitions(tachiai_openvpn PRIVATE TACHIAI_OPENVPN_TEST_HOOKS)
  target_compile_definitions(tachiai_openvpn_test PRIVATE TACHIAI_OPENVPN_TEST_HOOKS)
  target_link_libraries(tachiai_openvpn_test PRIVATE tachiai_openvpn)
endif()
